> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新建 `@vendure/affiliate-plugin`，实现一级分销返利：推广员生成推广码、新用户首绑终身、好友下单送达返佣（商品级可配费率+可配负载方）、退款回滚、待提现+人工打款兑付。

**Tech Stack:** Node + NestJS + TypeORM + Vendure core 3.6 + shop-plugin + vitest(SQLite via sqljs)。

## 契约/命名速查（跨任务一致）

- 实体继承 `VendureEntity implements ChannelAware`（import { Channel, ChannelAware, DeepPartial, VendureEntity } from '@vendure/core'），`constructor(input?: DeepPartial<T>){super(input);}`；`channelId` 普通列 + `@ManyToMany(()=>Channel)+@JoinTable() channels`；time/Duration/枚举列显式 `@Column({ type: 'varchar'/'int'/'datetime' })`（勿联合类型反射 Object）；幂等/唯一用 `@Index({ unique: true })`。
- 插件 options：`export const AFFILIATE_PLUGIN_OPTIONS = Symbol(...)` + `static options` + `init(options?): typeof AffiliatePlugin`；providers `{ provide: AFFILIATE_PLUGIN_OPTIONS, useFactory: () => AffiliatePlugin.options }`。
- 双 schema：`affiliateTypeDefs` 常量（共享实体类型，字段与实体一致）在 adminSchema/shopSchema **各插一份**（阶段27 铁律，勿跨 admin/shop 引用）。
- 金额/费率单位：金额分(bigint)；费率千分比 int（`round(base*rate/1000)`）。
- `order` 事件关系不完整：service 内 `orderService.findOne(ctx, id, ['customer','lines','lines.productVariant','lines.productVariant.product','lines.productVariant.product.translations'])` 重载。
- 店归属/费率：OrderLine 无私货 productId（阶段22/24/28），须 `line.productVariant.productId → Product`；`shopId=Number(product.customFields.shopId)`；`rate` 取商品 `customFields.affiliateRate`(千分比) → 店铺级 → 全局默认。
- 店主域鉴权：`administratorService.findOneByUserId` → `getRepository(ctx, Shop).findOne({administratorId})` + `status==='active'`（直接仓储查 Shop，勿注入 shop.service 防 DI 环）。
- 构建：插件目录 `npm run build`（= `rimraf lib && tsc -p ./tsconfig.json`）；改实体 schema 后删 `e2e/__data__/*.sqlite` 重跑。
- e2e 运行：插件目录内 `$env:PACKAGE='affiliate-plugin'; npx vitest run --config vitest.config.mts e2e/affiliate.e2e-spec.ts`。

```ts
import { Inject, Injectable } from '@nestjs/common';
import {
    AdministratorService, Customer, EntityNotFoundError, ForbiddenError, ID, Order,
    OrderService, RequestContext, TransactionalConnection, UserInputError,
} from '@vendure/core';
import { Shop } from '@vendure/shop-plugin';

import { AFFILIATE_PLUGIN_OPTIONS, AffiliatePluginOptions } from './affiliate.options';
import { Affiliate } from './affiliate.entity';
import { AffiliateRelation } from './affiliate-relation.entity';
import { AffiliateCommissionEntry, LoadOn } from './affiliate-commission.entity';
import { AffiliateWithdrawal } from './affiliate-withdrawal.entity';

const CODE_CHARS = 'ABCDEFGHJKMNPQRSTUVWXYZ23456789';
const MS_PER_DAY = 86400000;

@Injectable()
export class AffiliateService {
    constructor(
        @Inject(AFFILIATE_PLUGIN_OPTIONS) private options: AffiliatePluginOptions,
        private connection: TransactionalConnection,
        private orderService: OrderService,
        private administratorService: AdministratorService,
    ) {}

    /** 店主域鉴权（复用阶段18）。 */
    async requireMyShop(ctx: RequestContext): Promise<Shop> {
        if (!ctx.activeUserId) throw new ForbiddenError();
        const admin = await this.administratorService.findOneByUserId(ctx, ctx.activeUserId);
        if (!admin || admin.id == null) throw new ForbiddenError();
        const shop = await this.connection.getRepository(ctx, Shop).findOne({ where: { administratorId: admin.id as number } } as any);
        if (!shop || shop.status !== 'active') throw new ForbiddenError();
        return shop;
    }

    /** 当前登录用户的 Customer（按 customer.user.id）。 */
    async customerOf(ctx: RequestContext): Promise<Customer | undefined> {
        if (!ctx.activeUserId) throw new ForbiddenError();
        return this.connection.getRepository(ctx, Customer).findOne({ where: { user: { id: ctx.activeUserId } } } as any);
    }

    private async genUniqueCode(ctx: RequestContext): Promise<string> {
        const repo = this.connection.getRepository(ctx, Affiliate);
        for (let i = 0; i < 10; i++) {
            const ts = Date.now().toString(36).toUpperCase();
            let rnd = '';
            for (let j = 0; j < 6; j++) rnd += CODE_CHARS[Math.floor(Math.random() * CODE_CHARS.length)];
            const code = `${ts}${rnd}`;
            const hit = await repo.findOne({ where: { code } });
            if (!hit) return code;
        }
        throw new UserInputError('Failed to generate unique affiliate code');
    }

    /** 申请成为推广员（可指定归属店可选）。 */
    async becomeAffiliate(ctx: RequestContext, shopId?: ID): Promise<Affiliate> {
        if (!ctx.activeUserId) throw new ForbiddenError();
        const repo = this.connection.getRepository(ctx, Affiliate);
        const existing = await repo.findOne({ where: { userId: ctx.activeUserId as number } });
        if (existing) throw new UserInputError('Already an affiliate');
        const code = await this.genUniqueCode(ctx);
        return repo.save(repo.create({
            channelId: ctx.channelId as number,
            userId: ctx.activeUserId as number,
            shopId: shopId ? Number(shopId) : null,
            code,
            status: 'active',
            totalCommission: 0,
            withdrawableCommission: 0,
        }));
    }

    /** 绑定关系：首绑终身（点击/输码），重复/自绑拦截。 */
    async bindRelation(ctx: RequestContext, code: string, bindSource: 'click' | 'code'): Promise<AffiliateRelation> {
        const customer = await this.customerOf(ctx);
        if (!customer) throw new ForbiddenError();
        const affRepo = this.connection.getRepository(ctx, Affiliate);
        const aff = await affRepo.findOne({ where: { code } });
        if (!aff) throw new UserInputError('Affiliate code not found');
        if (aff.userId === ctx.activeUserId) throw new UserInputError('Cannot bind to yourself');
        const relRepo = this.connection.getRepository(ctx, AffiliateRelation);
        const existing = await relRepo.findOne({ where: { customerId: customer.id as number } });
        if (existing) throw new UserInputError('Already bound to an affiliate');
        return relRepo.save(relRepo.create({
            channelId: ctx.channelId as number,
            affiliateId: aff.id as number,
            customerId: customer.id as number,
            bindSource,
            boundAt: new Date(),
        }));
    }

    /** 幂等生成佣金：订单送达(Delivered)。 */
    async getOrCreateCommissions(ctx: RequestContext, order: Order): Promise<AffiliateCommissionEntry[]> {
        const orderId = order.id as number;
        const repo = this.connection.getRepository(ctx, AffiliateCommissionEntry);
        const existing = await repo.find({ where: { orderId } });
        if (existing.length > 0) return existing;
        const fresh = await this.orderService.findOne(ctx, orderId, [
            'customer', 'lines', 'lines.productVariant', 'lines.productVariant.product',
            'lines.productVariant.product.translations', 'lines.productVariant.translations',
        ]);
        if (!fresh?.customer || !fresh.lines?.length) return [];
        // 该单顾客是否绑定推广员
        const rel = await this.connection.getRepository(ctx, AffiliateRelation).findOne({
            where: { customerId: fresh.customer.id as number, channelId: ctx.channelId as number },
        });
        if (!rel) return [];
        const aff = await this.connection.getRepository(ctx, Affiliate).findOne({ where: { id: rel.affiliateId } });
        if (!aff || aff.status !== 'active') return [];
        const defaultLoadOn: LoadOn = this.options.defaultLoadOn ?? 'platform';
        const created: AffiliateCommissionEntry[] = [];
        for (const line of fresh.lines ?? []) {
            const product = line.productVariant?.product;
            const cf = (product?.customFields ?? {}) as any;
            const shopIdVal = cf.shopId;
            if (shopIdVal == null) continue;
            const shopId = Number(shopIdVal);
            const rate = this.resolveRate(cf, this.options.defaultRate);
            const baseAmount = Number(line.taxPriceTotal ?? 0) || 0;
            if (baseAmount <= 0) continue;
            const commissionAmount = Math.round((baseAmount * rate) / 1000);
            if (commissionAmount <= 0) continue;
            const entry = repo.create({
                channelId: ctx.channelId as number, affiliateId: aff.id as number,
                customerId: rel.customerId, orderId, orderLineId: line.id as number,
                shopId, baseAmount, rate, commissionAmount, loadOn: defaultLoadOn, status: 'pending',
                withdrawalId: null,
            });
            const saved = await repo.save(entry);
            created.push(saved);
        }
        return created;
    }

    /** 费率优先级：商品 affiliateRate → 全局默认。 */
    private resolveRate(cf: any, defaultRate: number | undefined): number {
        const productRate = Number(cf.affiliateRate ?? 0);
        if (productRate > 0) return productRate;
        return defaultRate ?? 30;
    }

    /** 退款回滚：该单 pending → reversed，并扣回可提现余额。 */
    async rollbackCommissions(ctx: RequestContext, orderId: ID): Promise<number> {
        const repo = this.connection.getRepository(ctx, AffiliateCommissionEntry);
        const rows = await repo.find({ where: { orderId: Number(orderId), status: 'pending' } });
        const affRepo = this.connection.getRepository(ctx, Affiliate);
        let count = 0;
        for (const row of rows) {
            row.status = 'reversed';
            await repo.save(row);
            const aff = await affRepo.findOne({ where: { id: row.affiliateId } });
            if (aff) {
                aff.withdrawableCommission = Math.max(0, aff.withdrawableCommission - row.commissionAmount);
                await affRepo.save(aff);
            }
            count++;
        }
        return count;
    }

    /** 汇总可提现余额（重新计算，防漂移）。 */
    async reconcileWithdrawable(ctx: RequestContext, affiliateId: number): Promise<number> {
        const repo = this.connection.getRepository(ctx, AffiliateCommissionEntry);
        const pendingRows = await repo.find({ where: { affiliateId, status: 'pending' } });
        const sumCom = pendingRows.reduce((s, r) => s + r.commissionAmount, 0);
        const wd = this.connection.getRepository(ctx, AffiliateWithdrawal);
        const nonRejected = await wd.find({ where: { affiliateId, status: 'paid' } });
        const withdrawnPaid = nonRejected.reduce((s, r) => s + r.amount, 0);
        return Math.max(0, sumCom - withdrawnPaid);
    }

    /** C 端我的推广信息。 */
    async myAffiliate(ctx: RequestContext): Promise<Affiliate | undefined> {
        if (!ctx.activeUserId) throw new ForbiddenError();
        return this.connection.getRepository(ctx, Affiliate).findOne({ where: { userId: ctx.activeUserId as number } });
    }

    /** C 端我的佣金明细。 */
    async myCommissionEntries(ctx: RequestContext): Promise<AffiliateCommissionEntry[]> {
        const aff = await this.myAffiliate(ctx);
        if (!aff) return [];
        return this.connection.getRepository(ctx, AffiliateCommissionEntry).find({ where: { affiliateId: aff.id as number }, order: { createdAt: 'DESC' } as any });
    }

    /** C 端提现申请：校验余额充足，生成 pending，冻结余额。 */
    async requestWithdrawal(ctx: RequestContext, amount: number): Promise<AffiliateWithdrawal> {
        const aff = await this.myAffiliate(ctx);
        if (!aff) throw new ForbiddenError();
        const avail = await this.reconcileWithdrawable(ctx, aff.id as number);
        if (amount > avail) throw new UserInputError('Withdrawal amount exceeds available balance');
        const repo = this.connection.getRepository(ctx, AffiliateWithdrawal);
        return repo.save(repo.create({
            channelId: ctx.channelId as number, affiliateId: aff.id as number,
            amount, status: 'pending', paidAt: null, note: null,
        }));
    }
}
```

function mergeCustomFields<T extends { name: string }>(existing: T[] | undefined, additions: T[] | undefined): T[] {
    const names = new Set((existing ?? []).map(f => f.name));
    return [...(existing ?? []), ...(additions ?? []).filter(f => !names.has(f.name))];
}

        AffiliateService,
    ],
    entities: [Affiliate, AffiliateRelation, AffiliateCommissionEntry, AffiliateWithdrawal],
    adminApiExtensions: { schema: adminSchema, resolvers: [AffiliateAdminResolver] },
    shopApiExtensions: { schema: shopSchema, resolvers: [AffiliateShopResolver] },
    configuration: (config) => {
        config.customFields.Product = mergeCustomFields(config.customFields.Product, [AFFILIATE_RATE_CUSTOM_FIELD]);
        return config;
    },

})
export class AffiliatePlugin implements OnApplicationBootstrap {
    private static options: AffiliatePluginOptions = {};

    constructor(private service: AffiliateService, private eventBus: EventBus) {}

    static init(options?: AffiliatePluginOptions): typeof AffiliatePlugin {
        AffiliatePlugin.options = options ?? {};
        return AffiliatePlugin;
    }

    onApplicationBootstrap(): void {
        // 送达 → 生成佣金
        this.eventBus.ofType(OrderStateTransitionEvent)
            .pipe(filter(e => e.toState === 'Delivered'))
            .subscribe(e => {
                const orderId = e.order?.id;
                if (orderId == null) return;
                this.service.getOrCreateCommissions(e.ctx, e.order).catch(err => Logger.error(err?.message, loggerCtx));
            });
        // 退款 Settled → 回滚
        this.eventBus.ofType(RefundStateTransitionEvent)
            .pipe(filter(e => e.toState === 'Settled'))
            .subscribe(e => {
                const orderId = e.order?.id;
                if (orderId == null) return;
                this.service.rollbackCommissions(e.ctx, orderId).catch(err => Logger.error(err?.message, loggerCtx));
            });

async payWithdrawalSafe(ctx: RequestContext, withdrawalId: ID): Promise<AffiliateWithdrawal> {
    await this.requireMyShop(ctx); // 店主域；平台可用 UpdateSettings
    const repo = this.connection.getRepository(ctx, AffiliateWithdrawal);
    const w = await repo.findOne({ where: { id: Number(withdrawalId) } });
    if (!w) throw new EntityNotFoundError('AffiliateWithdrawal', withdrawalId);
    if (w.status === 'paid') return w;
    w.status = 'paid';
    w.paidAt = new Date();
    return repo.save(w);
}

async rejectWithdrawalSafe(ctx: RequestContext, withdrawalId: ID): Promise<AffiliateWithdrawal> {
    await this.requireMyShop(ctx);
    const repo = this.connection.getRepository(ctx, AffiliateWithdrawal);
    const w = await repo.findOne({ where: { id: Number(withdrawalId) } });
    if (!w) throw new EntityNotFoundError('AffiliateWithdrawal', withdrawalId);
    if (w.status === 'rejected') return w;
    if (w.status === 'pending') {
        // 驳回回放余额
        const aff = await this.connection.getRepository(ctx, Affiliate).findOne({ where: { id: w.affiliateId } });
        if (aff) {
            aff.withdrawableCommission = await this.reconcileWithdrawable(ctx, aff.id as number);
            await this.connection.getRepository(ctx, Affiliate).save(aff);
        }
    }
    w.status = 'rejected';
    return repo.save(w);
}
